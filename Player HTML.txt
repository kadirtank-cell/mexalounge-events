        <script>
            function updatePlayer() {
                let cacheBuster = '_t=' + new Date().getTime();
                let apiUrl = 'https://radyohost.ugnhosting.com/cp/get_info.php?p=8018&' + cacheBuster;

                fetch(apiUrl)
                    .then(response => response.json())
                    .then(data => {
                        let trackTitle = data.title ? data.title : "Mexalounge Radio (Live)";
                        let artUrl = data.art ? data.art : "";

                        let randomBonus = Math.floor(Math.random() * (150 - 30 + 1)) + 30;
                        let rawListeners = (data.listeners !== undefined && !isNaN(data.listeners)) ? parseInt(data.listeners) : 0;
                        let fakeListeners = rawListeners + randomBonus;

                        let historyHtml = '';
                        if(data.history && Array.isArray(data.history) && data.history.length > 0) {
                            data.history.forEach((track) => {
                                historyHtml += `<li>${track}</li>`;
                            });
                        } else {
                            historyHtml = "<li>No recent tracks found.</li>";
                        }

                        let widgets = document.querySelectorAll('.player-widget');
                        widgets.forEach(widget => {
                            let nowPlayingEl = widget.querySelector('.now-playing');
                            let artEl = widget.querySelector('.art');
                            let listenersEl = widget.querySelector('.listeners');
                            let historyEl = widget.querySelector('.history-list');

                            if (nowPlayingEl) nowPlayingEl.innerText = "Now Playing: " + trackTitle;
                            if (artEl) {
                                if (artUrl) {
                                    artEl.src = artUrl;
                                    artEl.style.display = "block";
                                } else {
                                    artEl.style.display = "none";
                                }
                            }
                            if (listenersEl) listenersEl.innerText = "Active Listeners: " + fakeListeners;
                            if (historyEl) historyEl.innerHTML = historyHtml;
                        });
                    })
                    .catch(error => {
                        console.log('API fetch error:', error);
                    });
            }

            setInterval(updatePlayer, 8000);
            updatePlayer();
</script>
